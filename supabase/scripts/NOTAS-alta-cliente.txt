Agregar un nuevo cliente: 

-------------------------------------------------------------------------------------------

DO $$ 
DECLARE
    -- ==========================================
    -- 📝 CONFIGURACIÓN: CAMBIA ESTOS DATOS
    -- ==========================================
    v_customer_name    TEXT := 'Nombre del Cliente';   -- Nombre de la empresa
    v_customer_slug    TEXT := 'nombre-cliente';      -- Slug para la URL (sin espacios)
    v_branch_name      TEXT := 'Sucursal Central';    -- Nombre de la primera sucursal
    v_user_email       TEXT := 'admin@cliente.com';   -- Email del usuario (debe existir en Auth)
    v_auth_user_id     UUID := 'PEGAR-AQUÍ-EL-ID-DE-SUPABASE-AUTH'; -- ID que copias de Auth
    
    -- Variables internas (no tocar)
    v_new_tenant_id    UUID;
    v_new_branch_id    UUID;
BEGIN
    -- 1. Crear el Tenant
    INSERT INTO tenants (name, slug, email)
    VALUES (v_customer_name, v_customer_slug, v_user_email)
    RETURNING id INTO v_new_tenant_id;

    -- 2. Crear la Sucursal vinculada al Tenant
    INSERT INTO branches (tenant_id, name, location)
    VALUES (v_new_tenant_id, v_branch_name, 'Ubicación pendiente')
    RETURNING id INTO v_new_branch_id;

    -- 3. Crear el Perfil vinculado a la Sucursal y al Usuario de Auth
    -- Nota: Si el usuario ya tenía un perfil, esto lo actualizará (UPSERT)
    INSERT INTO profiles (id, tenant_id, branch_id, email, role)
    VALUES (v_auth_user_id, v_new_tenant_id, v_new_branch_id, v_user_email, 'ADMIN')
    ON CONFLICT (id) DO UPDATE 
    SET tenant_id = EXCLUDED.tenant_id, 
        branch_id = EXCLUDED.branch_id, 
        role = 'ADMIN';

    -- 4. Mostrar feedback en la consola de Supabase
    RAISE NOTICE '✅ Cliente creado con ID: %', v_new_tenant_id;
    RAISE NOTICE '✅ Sucursal creada con ID: %', v_new_branch_id;
    RAISE NOTICE '✅ Perfil vinculado correctamente.';

END $$;

--------------------------------------------------------------------------------------------------

Crea el usuario en Supabase Auth: Ve a Authentication -> Users -> Add User. Copia el User UID.

Pega el script en el SQL Editor:

Cambia v_customer_name.
Cambia v_customer_slug.
Pega el User UID en v_auth_user_id.

Ejecuta (Run).
